(import (only-in :gerbil/runtime/loader set-load-path!) :gerbil/expander (only-in :gerbil/runtime/init gerbil-load-expander!))
(export main)
(def (main . paths)
  (unless (null? paths) (set-load-path! paths))
  (display "MAIN before-expander\n") (force-output)
  (gerbil-load-expander!)
  (display "MAIN after-expander\n") (force-output)
  (parameterize ((current-expander-context (make-top-context)))
    (eval-syntax '(import :gerbil/core))
    (display (eval-syntax '(let ((x 21)) (+ x x))))
    (newline) (force-output)))
