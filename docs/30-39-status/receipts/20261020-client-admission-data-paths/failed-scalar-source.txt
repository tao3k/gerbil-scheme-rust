"""Same-VM scalar cycle calibration; native Scheme executes in both API paths."""
import json,math,os,statistics,time
import gerbil_ledger
from benchmark_host import host_snapshot

def percentile(values,p=.99):return sorted(values)[max(0,math.ceil(len(values)*p)-1)]
library=gerbil_ledger.open_library();states=library.vm_states
samples={"ordinary":[],"direct-batch":[]}
try:
    before=host_snapshot()
    for index in range(544):
        for arm in (["ordinary","direct-batch"] if index%4 in (0,3) else ["direct-batch","ordinary"]):
            started=time.perf_counter_ns()
            if arm=="ordinary":
                root=library.session(index%4).call(("echo",index))
                value=root.to_python();root.close()
            else:
                roots=library.call_many([("echo",index)],dispatch="direct",admission_window=16)
                value=library.project_many(roots,release=True)[0]
            elapsed=time.perf_counter_ns()-started
            assert value==index
            if index>=32:samples[arm].append(elapsed)
    library.set_owner_profiling(True)
    for index in range(32):
        root=library.session(index%4).call(("echo",index));assert root.to_python()==index;root.close()
    trace=json.loads(library.owner_profile());library.set_owner_profiling(False)
    assert not trace["dropped"]
    phase={}
    for kind in ["call","inspect","release"]:
        rows=[r for r in trace["spans"] if r["kind"]==kind]
        phase[kind]=dict(native_wall_ns_p50=statistics.median(r["end_ns"]-r["start_ns"] for r in rows),native_wall_ns_p99=percentile([r["end_ns"]-r["start_ns"] for r in rows]),owner_wait_ns_p99=percentile([r["start_ns"]-r["queued_ns"] for r in rows]),input_stage_ns_p50=statistics.median(r["stages_ns"][0] for r in rows) if kind=="call" else None,scheme_boundary_ns_p50=statistics.median(r["stages_ns"][1]-r["stages_ns"][0] for r in rows) if kind=="call" else None))
    path=library.runtime_directory;library.close()
    assert library.shutdown_clean and library.shutdown_roots==[0]*4 and not os.path.exists(path)
    print(json.dumps(dict(kind="scalar-cycle-calibration",states=states,one_outstanding=True,rounds=512,results={arm:dict(p50_ns=statistics.median(v),p99_ns=percentile(v),samples=v) for arm,v in samples.items()},traced_native_phases=phase,roots=library.shutdown_roots,host_before=before,host_after=host_snapshot())))
finally:library.close()
