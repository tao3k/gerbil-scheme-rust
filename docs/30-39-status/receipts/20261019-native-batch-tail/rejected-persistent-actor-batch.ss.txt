;;; Fixed-owner mailbox actors. No Rust scheduler or foreign Scheme references.
(import :std/sync/completion)
(export make-gerbil-rs-actor-batch gerbil-rs-actor-batch! gerbil-rs-actor-batch-close!)

(defstruct actor-batch (workers busy closed coordinator completion))
(defstruct actor-request (input quantum))
(defstruct actor-turn (reply input output quantum width))

(def (actor-turn-run! handler worker turn)
  (let loop ((index worker) (done 0) (ok #t))
    (if (>= index (vector-length (actor-turn-input turn))) ok
      (let (success
            (with-catch (lambda (_) #f)
              (lambda ()
                (vector-set! (actor-turn-output turn) index
                  (handler worker (vector-ref (actor-turn-input turn) index)))
                #t)))
        (when (and (positive? (actor-turn-quantum turn))
                   (zero? (modulo (1+ done) (actor-turn-quantum turn))))
          (thread-yield!))
        (loop (+ index (actor-turn-width turn)) (1+ done) (and ok success))))))
(def (actor-batch-worker handler worker)
  (let loop ()
    (let (turn (thread-receive))
      (unless (eq? turn 'stop)
        (thread-send (actor-turn-reply turn) (actor-turn-run! handler worker turn))
        (loop)))))
(def (make-gerbil-rs-actor-batch handler count)
  (unless (and (procedure? handler) (integer? count) (<= 1 count 64))
    (error "invalid Scheme actor service"))
  (let (workers (make-vector count))
    (let loop ((worker 0))
      (when (< worker count)
        (vector-set! workers worker (spawn (lambda () (actor-batch-worker handler worker))))
        (loop (1+ worker))))
    (let (service (make-actor-batch workers #f #f #f (make-completion)))
      (set! (actor-batch-coordinator service) (spawn (lambda () (actor-batch-coordinate service))))
      service)))
(def (actor-batch-check! service)
  (when (or (actor-batch-closed service) (actor-batch-busy service))
    (error "Scheme actor service closed or reentered")))
(def (actor-batch-execute! service input quantum)
  (let* ((workers (actor-batch-workers service)) (count (vector-length workers))
         (output (make-vector (vector-length input)))
         (turn (make-actor-turn (current-thread) input output quantum count)))
    (vector-for-each (lambda (worker) (thread-send worker turn)) workers)
    ;; Every actor drains every assigned item even after a handler exception.
    ;; No turn or acknowledgement may survive the synchronous foreign entry.
    (let loop ((left count) (ok #t))
      (if (zero? left)
        (if ok output (error "Scheme actor batch handler failed"))
        (let (reply (thread-receive)) (loop (1- left) (and ok reply)))))))
(def (actor-batch-coordinate service)
  (let loop ()
    (let (request (thread-receive))
      (unless (eq? request 'stop)
        (with-catch
          (lambda (exn) (completion-error! (actor-batch-completion service) exn))
          (lambda ()
            (let (output (actor-batch-execute! service
                           (actor-request-input request) (actor-request-quantum request)))
              (completion-post! (actor-batch-completion service) output))))
        (loop)))))
(def (gerbil-rs-actor-batch! service envelope)
  (actor-batch-check! service)
  (unless (and (vector? envelope) (= (vector-length envelope) 2)
               (s64vector? (vector-ref envelope 0))
               (= (s64vector-length (vector-ref envelope 0)) 1)
               (vector? (vector-ref envelope 1)))
    (error "invalid native actor batch envelope"))
  (let ((quantum (s64vector-ref (vector-ref envelope 0) 0))
        (input (vector-ref envelope 1)))
    (unless (and (<= 0 quantum 256) (<= 1 (vector-length input) 256))
      (error "native actor batch bound"))
    (dynamic-wind
      (lambda () (set! (actor-batch-busy service) #t))
      ;; Reuse one private Scheme coordinator. The standard Completion keeps
      ;; acknowledgements out of the application's foreign-entry mailbox.
      (lambda ()
        (thread-send (actor-batch-coordinator service) (make-actor-request input quantum))
        (completion-wait! (actor-batch-completion service)))
      (lambda ()
        ;; Do not retain the last result/exception in a persistent completion.
        (completion-reset! (actor-batch-completion service))
        (set! (actor-batch-busy service) #f)))))
(def (gerbil-rs-actor-batch-close! service)
  (actor-batch-check! service)
  (thread-send (actor-batch-coordinator service) 'stop)
  (thread-join! (actor-batch-coordinator service))
  (set! (actor-batch-coordinator service) #f)
  (vector-for-each (lambda (worker) (thread-send worker 'stop)) (actor-batch-workers service))
  (vector-for-each thread-join! (actor-batch-workers service))
  (set! (actor-batch-workers service) #())
  (set! (actor-batch-closed service) #t))
